아래 전체 코드를 Google Apps Script 편집기(Code.gs)에 그대로 붙여넣고 저장한 뒤, [배포 → 배포 관리 → ✏️ 수정 → 새 버전 → 배포]를 눌러 주세요. (기존 웹 앱 URL은 그대로 유지됩니다!)

• 기존 플레이 로그 시트(첫 번째 탭)는 그대로 유지되며, 통합 마스터 탭(PlayerProgress)이 자동 생성되어 기기 간 동기화 및 관리자 조회·수정이 동작합니다.

var ADMIN_PASSWORD = '3412';
var PROGRESS_SHEET_NAME = 'PlayerProgress';
var PROGRESS_HEADERS = [
  'playerId', 'updatedAt', 'epSummary', 'notebookCount', 'missionCount',
  'footstepsSummary', 'episodesJson', 'notebookJson', 'footstepsJson',
  'revision', 'adminOverrideRev', 'lastModifiedBy'
];

function getLogSheet_(ss) {
  var sheets = ss.getSheets();
  for (var i = 0; i < sheets.length; i++) {
    if (sheets[i].getName() !== PROGRESS_SHEET_NAME) return sheets[i];
  }
  return sheets[0];
}

function getOrCreateProgressSheet_(ss) {
  var sheet = ss.getSheetByName(PROGRESS_SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(PROGRESS_SHEET_NAME);
    sheet.appendRow(PROGRESS_HEADERS);
    sheet.setFrozenRows(1);
    sheet.getRange(1, 1, 1, PROGRESS_HEADERS.length)
      .setFontWeight('bold')
      .setBackground('#1e293b')
      .setFontColor('#fde047');
    sheet.setColumnWidths(1, 6, 150);
    sheet.setColumnWidths(7, 3, 240);
  } else if (sheet.getLastRow() === 0) {
    sheet.appendRow(PROGRESS_HEADERS);
    sheet.setFrozenRows(1);
  }
  return sheet;
}

function kstNow_() {
  return Utilities.formatDate(new Date(), 'Asia/Seoul', 'yyyy-MM-dd HH:mm:ss');
}

function parseJsonSafe_(raw, fallback) {
  try {
    if (!raw) return fallback;
    var parsed = JSON.parse(String(raw));
    return parsed && typeof parsed === 'object' ? parsed : fallback;
  } catch (err) {
    return fallback;
  }
}

function rowToProfile_(row) {
  return {
    playerId: String(row[0] || '').trim(),
    updatedAt: String(row[1] || ''),
    episodes: parseJsonSafe_(row[6], { progress: {}, certs: {} }),
    notebook: parseJsonSafe_(row[7], { collected: [], completedMissions: [], clearedStages: {}, postResurrection: false, ascensionSeen: false }),
    footsteps: parseJsonSafe_(row[8], {}),
    revision: Number(row[9] || 0),
    adminOverrideRev: Number(row[10] || 0),
    lastModifiedBy: String(row[11] || 'player')
  };
}

function collectOrderedKeys_(baseOrder, objList) {
  var out = [];
  var seen = {};
  var i, j, k;
  for (i = 0; i < baseOrder.length; i++) {
    k = String(baseOrder[i] || '').trim();
    if (k && !seen[k]) { seen[k] = true; out.push(k); }
  }
  var extra = [];
  var list = Array.isArray(objList) ? objList : [];
  for (i = 0; i < list.length; i++) {
    var obj = list[i];
    if (!obj || typeof obj !== 'object') continue;
    var keys = Object.keys(obj);
    for (j = 0; j < keys.length; j++) {
      k = String(keys[j] || '').trim();
      if (k && !seen[k]) { seen[k] = true; extra.push(k); }
    }
  }
  extra.sort();
  return out.concat(extra);
}

function buildSummaries_(profile) {
  var epLabels = { noah: 'EP1', exodus: 'EP2', david: 'EP3', miracle: 'EP4' };
  var eps = profile.episodes || {};
  var prog = eps.progress || {};
  var certs = eps.certs || {};
  var epOrder = collectOrderedKeys_(['noah', 'exodus', 'david', 'miracle'], [prog, certs]);
  var epParts = [];
  for (var i = 0; i < epOrder.length; i++) {
    var k = epOrder[i];
    var epLbl = epLabels[k] || String(k).toUpperCase();
    var p = prog[k] || {};
    var c = certs[k];
    if (c && c.code) {
      epParts.push(epLbl + ':완주(' + c.code + ')');
    } else if (p.completed || p.certIssued) {
      epParts.push(epLbl + ':완주');
    } else if (Number(p.maxClearedStage || 0) > 0) {
      epParts.push(epLbl + ':ST' + p.maxClearedStage);
    }
  }
  var nb = profile.notebook || {};
  var nbLen = Array.isArray(nb.collected) ? nb.collected.length : 0;
  var msLen = Array.isArray(nb.completedMissions) ? nb.completedMissions.length : 0;
  var fp = profile.footsteps || {};
  var fpOrder = collectOrderedKeys_(['fp_1', 'fp_2', 'fp_3', 'fp_4', 'fp_5'], [fp]);
  var fpCodes = [];
  for (var j = 0; j < fpOrder.length; j++) {
    var fk = fpOrder[j];
    if (fp[fk] && fp[fk].code) fpCodes.push(fp[fk].code);
  }
  return {
    epSummary: epParts.length ? epParts.join(' | ') : '-',
    notebookCount: nbLen,
    missionCount: msLen,
    footstepsSummary: fpCodes.length ? fpCodes.join(', ') : '-'
  };
}

function uniqueStrings_(a, b) {
  var map = {};
  var out = [];
  var arrA = Array.isArray(a) ? a : [];
  var arrB = Array.isArray(b) ? b : [];
  var combined = arrA.concat(arrB);
  for (var i = 0; i < combined.length; i++) {
    var v = String(combined[i] || '').trim();
    if (v && !map[v]) {
      map[v] = true;
      out.push(v);
    }
  }
  return out;
}

function mergeProfilesUnion_(existing, incoming) {
  var eEp = (existing && existing.episodes) || {};
  var iEp = (incoming && incoming.episodes) || {};
  var epOrder = collectOrderedKeys_(['noah', 'exodus', 'david', 'miracle'], [eEp.progress, iEp.progress, eEp.certs, iEp.certs]);
  var mergedProg = {};
  var mergedCerts = {};
  for (var i = 0; i < epOrder.length; i++) {
    var ep = epOrder[i];
    var lp = (eEp.progress && eEp.progress[ep]) || {};
    var rp = (iEp.progress && iEp.progress[ep]) || {};
    var maxSt = Math.max(Number(lp.maxClearedStage || 0), Number(rp.maxClearedStage || 0));
    mergedProg[ep] = {
      maxClearedStage: maxSt,
      completed: Boolean(lp.completed || rp.completed),
      certIssued: Boolean(lp.certIssued || rp.certIssued),
      attempts: Math.max(Number(lp.attempts || 0), Number(rp.attempts || 0))
    };
    var lc = eEp.certs && eEp.certs[ep];
    var rc = iEp.certs && iEp.certs[ep];
    if (lc || rc) {
      var baseC = lc || rc;
      var modes = uniqueStrings_(lc && lc.clearedModes, rc && rc.clearedModes);
      mergedCerts[ep] = {
        epId: ep,
        code: String((lc && lc.code && lc.code !== '-') ? lc.code : ((rc && rc.code) || '')),
        playerName: String((rc && rc.playerName) || (lc && lc.playerName) || incoming.playerId || ''),
        issuedAt: String((lc && lc.issuedAt) || (rc && rc.issuedAt) || ''),
        updatedAt: String((rc && rc.updatedAt) || (lc && lc.updatedAt) || ''),
        clearedModes: modes
      };
      if (!mergedCerts[ep].code) delete mergedCerts[ep];
    }
  }

  var eNb = (existing && existing.notebook) || {};
  var iNb = (incoming && incoming.notebook) || {};
  var mergedStages = {};
  for (var st = 1; st <= 4; st++) {
    mergedStages[st] = Boolean((eNb.clearedStages && eNb.clearedStages[st]) || (iNb.clearedStages && iNb.clearedStages[st]));
  }
  var mergedNb = {
    collected: uniqueStrings_(eNb.collected, iNb.collected),
    completedMissions: uniqueStrings_(eNb.completedMissions, iNb.completedMissions),
    clearedStages: mergedStages,
    postResurrection: Boolean(eNb.postResurrection || iNb.postResurrection),
    ascensionSeen: Boolean(eNb.ascensionSeen || iNb.ascensionSeen)
  };

  var eFp = (existing && existing.footsteps) || {};
  var iFp = (incoming && incoming.footsteps) || {};
  var fpOrder = collectOrderedKeys_(['fp_1', 'fp_2', 'fp_3', 'fp_4', 'fp_5'], [eFp, iFp]);
  var mergedFp = {};
  for (var f = 0; f < fpOrder.length; f++) {
    var fk = fpOrder[f];
    var lf = eFp[fk];
    var rf = iFp[fk];
    if (lf && lf.code) mergedFp[fk] = lf;
    else if (rf && rf.code) mergedFp[fk] = rf;
  }

  return {
    playerId: String((incoming && incoming.playerId) || (existing && existing.playerId) || '').trim(),
    updatedAt: kstNow_(),
    episodes: { progress: mergedProg, certs: mergedCerts },
    notebook: mergedNb,
    footsteps: mergedFp
  };
}

function findProgressRowIndex_(sheet, playerId) {
  var norm = String(playerId || '').trim().toLowerCase();
  if (!norm || sheet.getLastRow() < 2) return -1;
  var values = sheet.getRange(2, 1, sheet.getLastRow() - 1, 1).getValues();
  for (var i = 0; i < values.length; i++) {
    if (String(values[i][0] || '').trim().toLowerCase() === norm) {
      return i + 2;
    }
  }
  return -1;
}

function writeProfileRow_(sheet, rowIndex, profile) {
  var s = buildSummaries_(profile);
  var rowValues = [
    profile.playerId,
    profile.updatedAt,
    s.epSummary,
    s.notebookCount,
    s.missionCount,
    s.footstepsSummary,
    JSON.stringify(profile.episodes || { progress: {}, certs: {} }),
    JSON.stringify(profile.notebook || {}),
    JSON.stringify(profile.footsteps || {}),
    Number(profile.revision || 1),
    Number(profile.adminOverrideRev || 0),
    String(profile.lastModifiedBy || 'player')
  ];
  if (rowIndex >= 2) {
    sheet.getRange(rowIndex, 1, 1, rowValues.length).setValues([rowValues]);
  } else {
    sheet.appendRow(rowValues);
  }
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var logSheet = getLogSheet_(ss);
  var progSheet = getOrCreateProgressSheet_(ss);
  var params = (e && e.parameter) || {};
  var mode = String(params.mode || '').trim();
  var filterName = String(params.name || '').trim().toLowerCase();

  var logRows = logSheet.getDataRange().getValues();
  var list = [];
  for (var i = 1; i < logRows.length; i++) {
    var r = logRows[i];
    var rName = String(r[1] || '');
    if (mode === 'player' && filterName && rName.trim().toLowerCase() !== filterName) continue;
    list.push({
      time: r[0],
      name: rName,
      mode: r[2],
      episode: r[3],
      stage: r[4],
      status: r[5],
      playCount: r[6],
      certCode: r[7]
    });
  }

  var progRows = progSheet.getDataRange().getValues();
  if (mode === 'player') {
    var foundProfile = null;
    for (var j = 1; j < progRows.length; j++) {
      if (String(progRows[j][0] || '').trim().toLowerCase() === filterName) {
        foundProfile = rowToProfile_(progRows[j]);
        break;
      }
    }
    return ContentService.createTextOutput(JSON.stringify({
      ok: true,
      profileSupported: true,
      list: list,
      profile: foundProfile
    })).setMimeType(ContentService.MimeType.JSON);
  }

  var profiles = [];
  for (var k = 1; k < progRows.length; k++) {
    if (String(progRows[k][0] || '').trim()) {
      profiles.push(rowToProfile_(progRows[k]));
    }
  }
  return ContentService.createTextOutput(JSON.stringify({
    ok: true,
    profileSupported: true,
    list: list,
    profiles: profiles
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var logSheet = getLogSheet_(ss);
    var body = JSON.parse(e.postData.contents || '{}');

    // 1) 플레이어 통합 진행도 동기화 (합집합 병합 + 관리자 우선 제어)
    if (body.action === 'sync_profile') {
      var progSheet = getOrCreateProgressSheet_(ss);
      var incoming = body.profile || {};
      var pid = String(body.playerId || incoming.playerId || '').trim();
      if (!pid) {
        return ContentService.createTextOutput(JSON.stringify({ ok: false, error: 'missing_playerId' }))
          .setMimeType(ContentService.MimeType.JSON);
      }
      incoming.playerId = pid;
      var rowIndex = findProgressRowIndex_(progSheet, pid);
      var existing = null;
      if (rowIndex >= 2) {
        existing = rowToProfile_(progSheet.getRange(rowIndex, 1, 1, PROGRESS_HEADERS.length).getValues()[0]);
        var clientAppliedAdminRev = Number(body.appliedAdminRev || 0);
        if (Number(existing.adminOverrideRev || 0) > clientAppliedAdminRev) {
          return ContentService.createTextOutput(JSON.stringify({
            ok: true,
            adminOverride: true,
            profile: existing
          })).setMimeType(ContentService.MimeType.JSON);
        }
      }
      var merged = mergeProfilesUnion_(existing, incoming);
      merged.revision = (existing ? Number(existing.revision || 0) : 0) + 1;
      merged.adminOverrideRev = existing ? Number(existing.adminOverrideRev || 0) : 0;
      merged.lastModifiedBy = 'player';
      writeProfileRow_(progSheet, rowIndex, merged);
      return ContentService.createTextOutput(JSON.stringify({
        ok: true,
        revision: merged.revision,
        adminOverrideRev: merged.adminOverrideRev,
        profile: merged
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // 2) 관리자 상세 수정 / 신규 등록
    if (body.action === 'admin_upsert_profile') {
      if (String(body.password) !== ADMIN_PASSWORD) {
        return ContentService.createTextOutput(JSON.stringify({ ok: false, error: 'unauthorized' }))
          .setMimeType(ContentService.MimeType.JSON);
      }
      var progSheetAdmin = getOrCreateProgressSheet_(ss);
      var adminProf = body.profile || {};
      var targetPid = String(body.playerId || adminProf.playerId || '').trim();
      if (!targetPid) {
        return ContentService.createTextOutput(JSON.stringify({ ok: false, error: 'missing_playerId' }))
          .setMimeType(ContentService.MimeType.JSON);
      }
      var adminRowIdx = findProgressRowIndex_(progSheetAdmin, targetPid);
      var prevRev = 0;
      if (adminRowIdx >= 2) {
        prevRev = Number(progSheetAdmin.getRange(adminRowIdx, 10).getValue() || 0);
      }
      var nextRev = prevRev + 1;
      adminProf.playerId = targetPid;
      adminProf.updatedAt = kstNow_();
      adminProf.revision = nextRev;
      adminProf.adminOverrideRev = nextRev;
      adminProf.lastModifiedBy = 'admin';
      writeProfileRow_(progSheetAdmin, adminRowIdx, adminProf);
      return ContentService.createTextOutput(JSON.stringify({
        ok: true,
        revision: nextRev,
        adminOverrideRev: nextRev,
        profile: adminProf
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // 3) 관리자 통합 프로필 삭제
    if (body.action === 'admin_delete_profile') {
      if (String(body.password) !== ADMIN_PASSWORD) {
        return ContentService.createTextOutput(JSON.stringify({ ok: false, error: 'unauthorized' }))
          .setMimeType(ContentService.MimeType.JSON);
      }
      var progSheetDel = getOrCreateProgressSheet_(ss);
      var delIdx = findProgressRowIndex_(progSheetDel, body.playerId || body.name);
      var deletedProf = 0;
      if (delIdx >= 2) {
        progSheetDel.deleteRow(delIdx);
        deletedProf = 1;
      }
      return ContentService.createTextOutput(JSON.stringify({ ok: true, deleted: deletedProf }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // 4) 기존 플레이 로그 삭제 (+ 선택 시 통합 프로필 동시 삭제)
    if (body.action === 'delete' && String(body.password) === ADMIN_PASSWORD) {
      var rows = logSheet.getDataRange().getValues();
      var deleted = 0;
      for (var i = rows.length - 1; i >= 1; i--) {
        var rTime = String(rows[i][0] || '');
        var rName = String(rows[i][1] || '');
        var rStatus = String(rows[i][5] || '');
        if (body.deleteBy === 'name' && rName === String(body.name)) {
          logSheet.deleteRow(i + 1);
          deleted++;
        } else if (body.deleteBy === 'row' && rName === String(body.name) &&
                   (rTime === String(body.rawTime) || rStatus === String(body.status))) {
          logSheet.deleteRow(i + 1);
          deleted++;
          break;
        }
      }
      if (body.deleteBy === 'name' && body.includeProfile) {
        var ps = getOrCreateProgressSheet_(ss);
        var pIdx = findProgressRowIndex_(ps, body.name);
        if (pIdx >= 2) ps.deleteRow(pIdx);
      }
      return ContentService.createTextOutput(JSON.stringify({ ok: true, deleted: deleted }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // 5) 기존 플레이 이벤트 로그 추가
    var kstTime = body.time || kstNow_();
    logSheet.appendRow([kstTime, body.name, body.mode, body.episode, body.stage, body.status, body.playCount, body.certCode]);
    return ContentService.createTextOutput(JSON.stringify({ ok: true }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    try { lock.releaseLock(); } catch (err) {}
  }
}
