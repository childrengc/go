⚙️ 구글 스프레드시트 원본 영구 삭제 연동 방법

현재 관리자 페이지에서 삭제 버튼을 누르면 ① 구글 시트 서버로 삭제 요청을 전송하고, ② 즉시 관리자 화면·통계·CSV에서도 제거됩니다.
구글 스프레드시트 원본 시트의 행까지 영구적으로 지우려면, 사용 중인 Google Apps Script의 doPost(e) 상단에 아래 코드를 추가한 뒤 [배포 → 배포 관리 → 새 버전]으로 저장해 주세요. 

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var body = JSON.parse(e.postData.contents);

  // [추가] 관리자 기록 삭제 처리 (특정인 일괄 삭제 / 개별 행 삭제)
  if (body.action === 'delete' && String(body.password) === '0504') {
    var rows = sheet.getDataRange().getValues();
    var deleted = 0;
    for (var i = rows.length - 1; i >= 1; i--) {
      var rTime = String(rows[i][0] || '');
      var rName = String(rows[i][1] || '');
      var rStatus = String(rows[i][5] || '');
      if (body.deleteBy === 'name' && rName === String(body.name)) {
        sheet.deleteRow(i + 1);
        deleted++;
      } else if (body.deleteBy === 'row' && rName === String(body.name) &&
                 (rTime === String(body.rawTime) || rStatus === String(body.status))) {
        sheet.deleteRow(i + 1);
        deleted++;
        break;
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ ok: true, deleted: deleted }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // 기존 기록 추가 로직 (한국 시간 KST 포맷 저장)
  var kstTime = body.time || Utilities.formatDate(new Date(), 'Asia/Seoul', 'yyyy-MM-dd HH:mm:ss');
  sheet.appendRow([kstTime, body.name, body.mode, body.episode, body.stage, body.status, body.playCount, body.certCode]);
  return ContentService.createTextOutput(JSON.stringify({ ok: true }))
    .setMimeType(ContentService.MimeType.JSON);
}
